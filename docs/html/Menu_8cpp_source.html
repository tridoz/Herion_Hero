86void Menu::LoadButtonsMenuTypeConfiguration() {
87
88 try {
89 this->scale = JSONParser::graphics::GetScale();
90 this->filepath = JSONParser::menu_configuration::GetBackgroundImagePath();
91
92 this->start_y = JSONParser::menu_configuration::GetStartY();
93 this->button_y_offset = JSONParser::menu_configuration::GetButtonYOffset();
94 this->center_piece_offset = JSONParser::menu_configuration::GetCenterPieceOffset() * scale;
95 } catch ( HerionException::File::FileMalformedException& ex ) {
96 ex.UpdateStackTrace( GET_CONTEXT() );
97 throw;
98 }
99
100 try {
101 this->background = texture_manager->GetTexture(this->filepath );
102 } catch ( HerionException::File::FileException& ex ) {
103 ex.UpdateStackTrace( GET_CONTEXT() );
104 throw;
105 }
106
107 const int num_rows = JSONParser::menu_configuration::GetNumRows();
108 float current_y = start_y;
109
110 buttons.clear();
111 texts.clear();
112
113 for (int row_number = 0; row_number < num_rows; row_number++) {
114 const int starting_x = JSONParser::menu_configuration::GetRowStartingX(row_number) * scale;
115 const int button_x_offset = JSONParser::menu_configuration::GetRowButtonXOffset(row_number);
116 const int num_elements = JSONParser::menu_configuration::GetRowNumElements(row_number);
117
118 float cumulative_x = static_cast<float>(starting_x);
119 float row_height = 0.0f;
120
121 for (int element_number = 0; element_number < num_elements; element_number++) {
122 const JSONParser::menu_configuration::RowElementFields menu_element_characteristic = JSONParser::menu_configuration::GetRowElementFields(row_number, element_number);
123
124 std::string text = (menu_element_characteristic.type == "DYNAMIC_TEXT") ? GetText(menu_element_characteristic.text) : menu_element_characteristic.text;
125
126 std::vector<SDL_FRect> rects;
127 std::vector<Texture*> textures;
128
129 Texture* left_texture = nullptr;
130 Texture* center_texture = nullptr;
131 Texture* right_texture = nullptr;
132
133 try {
134 left_texture = texture_manager->GetTexture("Assets/Ui/Buttons/ButtonLeft2.png");
135 center_texture = texture_manager->GetTexture("Assets/Ui/Buttons/ButtonCenter2.png");
136 right_texture = texture_manager->GetTexture("Assets/Ui/Buttons/ButtonRight2.png");
137 } catch ( HerionException::File::FileNotFoundException &ex ) {
138 ex.UpdateStackTrace( GET_CONTEXT() );
139 throw;
140 }
141
142 float left_w, left_h, center_w, center_h, right_w, right_h;
143
144 SDL_GetTextureSize(left_texture->GetTexture(), &left_w, &left_h);
145 SDL_GetTextureSize(center_texture->GetTexture(), &center_w, &center_h);
146 SDL_GetTextureSize(right_texture->GetTexture(), &right_w, &right_h);
147
148 SDL_FRect left_rect = { cumulative_x, current_y, left_w * scale, left_h * scale };
149 rects.push_back(left_rect);
150 textures.push_back(left_texture);
151
152 float char_w = 20.0f * scale;
153 float text_total_w = text.size() * char_w;
154
155 SDL_FRect center_rect = { cumulative_x + left_rect.w,
156 current_y + center_piece_offset,
157 text_total_w,
158 center_h * scale };
159
160 rects.push_back(center_rect);
161 textures.push_back(center_texture);
162
163 SDL_FRect right_rect = { center_rect.x + center_rect.w, current_y, right_w * scale, right_h * scale };
164 rects.push_back(right_rect);
165 textures.push_back(right_texture);
166
167 float char_x = center_rect.x;
168
169 for (char c : text) {
170 Texture* char_tex = nullptr;
171
172 try {
173
174 if (std::isupper(c))
175 char_tex = texture_manager->GetTexture("Assets/Font/Game/UppercaseLetters/" + std::string(1, c) + ".png");
176 else if (std::islower(c))
177 char_tex = texture_manager->GetTexture("Assets/Font/Game/LowercaseLetters/" + std::string(1, c) + ".png");
178 else if (std::isdigit(c))
179 char_tex = texture_manager->GetTexture("Assets/Font/Game/Numbers/" + std::string(1, c) + ".png");
180 else if (isspecial(c))
181 char_tex = texture_manager->GetTexture("Assets/Font/Game/SpecialCharacters/" + GetNameOfSpecialChar(c) + ".png");
182 else if ( isspace(c))
183 char_tex = texture_manager->GetTexture("Assets/Font/Game/SpecialCharacters/space.png");
184
185 } catch ( HerionException::File::FileNotFoundException &ex ) {
186 ex.UpdateStackTrace( GET_CONTEXT() );
187 throw;
188 }
189
190 float cw, ch;
191
192 SDL_GetTextureSize(char_tex->GetTexture(), &cw, &ch);
193 SDL_FRect char_rect = {
194 char_x + (char_w - cw * scale) / 2.0f,
195 center_rect.y + (center_rect.h - ch * scale) / 2.0f,
196 cw * scale,
197 ch * scale
198 };
199 rects.push_back(char_rect);
200 textures.push_back(char_tex);
201
202 char_x += char_w;
203 }
204
205 row_height = std::max({ row_height, left_rect.h, center_rect.h, right_rect.h });
206
207 if (menu_element_characteristic.type == "BUTTON") {
208
209 Button* btn = new Button();
210 btn->SetRects(rects);
211 btn->SetTextures(textures);
212
213 if ( !buttons_functions.contains(menu_element_characteristic.action.value() ) ) {
214 THROW_FILE_NOT_FOUND( menu_element_characteristic.action.value() );
215 }
216
217 btn->SetOnClick(buttons_functions.at(menu_element_characteristic.action.value()));
218 buttons.emplace(menu_element_characteristic.id, btn);
219
220 } else if ( menu_element_characteristic.type.contains("TEXT") ) {
221 Text* txt = new Text();
222 txt->SetRects(rects);
223 txt->SetTextures(textures);
224 texts.emplace(menu_element_characteristic.id, txt);
225 }
226
227 cumulative_x += left_rect.w + center_rect.w + right_rect.w + button_x_offset;
228
229 }
230
231 current_y += row_height + button_y_offset;
232
233 }

404void Menu::CreateSubDirectories( Directory*& directory, const std::string& base_directory, int depth ) {
405
406 std::vector < std::string > cmd_output = CMD::multiline_exec( "ls " + base_directory + " | grep -v '^Font$' " );
407
408 if ( directory == nullptr ) {
409 directory = new Directory( depth, {}, {} );
410 }
411
412 if ( cmd_output.size() == 0) {
413 return;
414 }
415
416 for ( const auto& output_line : cmd_output ) {
417 Directory* dir = new Directory( depth+1, {}, {}, {} );
418 if ( !output_line.contains(".png") && !output_line.contains(".txt") ) {
419 directory->SubDirectory.emplace( std::make_pair(output_line, dir) );
420 dir->path = base_directory + output_line + "/";
421 CreateSubDirectories( dir, base_directory + output_line + "/", depth+1 );
422 } else if ( output_line.contains(".png") ) {
423 std::string file_name = base_directory + output_line;
424 file_name = file_name.substr(3,file_name.size()-3);
425 directory->Files.emplace_back( output_line, base_directory + output_line , this->texture_manager->GetTexture( file_name ) );
426 }
427
428 }
