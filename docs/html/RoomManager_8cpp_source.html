16RoomManager::~RoomManager() {
17 if (!spawn_room) return;
18
19 std::set<Node*> visited;
20
21 std::function<void(Node*)> FreeNode = [&](Node* node) {
22 if (!node || visited.count(node)) return;
23 visited.insert(node);
24
25 FreeNode(node->left);
26 FreeNode(node->right);
27 FreeNode(node->up);
28 FreeNode(node->down);
29
30 delete node->room;
31 delete node;
32 };
33
34 FreeNode(spawn_room);
35
36 spawn_room = nullptr;
37 current_room = nullptr;
38}

248void RoomManager::ResizeRoom() {
249 std::vector < std::vector< Tile* > > tiles = this->current_room->room->GetTiles();
250 int w, h;
251 try {
252 w = JSONParser::graphics::GetWidth();
253 h = JSONParser::graphics::GetHeight();
254 } catch ( HerionException::File::FileMalformedException& ex ) {
255 ex.UpdateStackTrace( GET_CONTEXT() );
256 throw;
257 }
258
259 for ( int x = 0 ; x < tiles.size() ; x++ ) {
260 for ( int y = 0 ; y < tiles[x].size() ; y++ ) {
261 float newW, newH;
262
263 newW = w / horizontal_tiles;
264 newH = h / vertical_tiles;
265
266 tiles[x][y] -> SetRect( y*newW, x*newH, newW, newH );
267
268 }
