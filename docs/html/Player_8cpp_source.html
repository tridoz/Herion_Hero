119void Player::LoadAnimation( const std::string &filepath ) {
120
121 try {
122 JSONParser::animations::SetConfigFile( filepath );
123 } catch ( HerionException::File::FileException& ex ) {
124 ex.UpdateStackTrace( GET_CONTEXT() );
125 throw;
126 }
127
128 int number_of_animations = JSONParser::animations::GetAnimationNumbers();
129
130 for ( int i = 0 ; i<number_of_animations ; i++ ) {
131 JSONParser::animations::AnimationElementsFields animation_characteristics = JSONParser::animations::GetAnimationElementsFields( i );
132 std::string animation_name = animation_characteristics.name;
133
134 animations.at( animation_name )->SetFrameRate( animation_characteristics.frame_number );
135 animations.at( animation_name )->SetAnimationFrameNUmber( animation_characteristics.frame_to_load );
136
137 for ( int j = 0 ; j<animation_characteristics.frame_to_load ; j++ ) {
138 Texture* texture = texture_manager->GetTexture( animation_characteristics.path + "frame" + std::to_string(j) + ".png" );
139 animations.at(animation_name)->AddFrameTexture( texture );
140 }
141 }

204void Player::Move(FacingDirection direction, float delta_time) {
205
206 const float distance = speed * delta_time;
207
208 bool is_moving = false;
209
210 if (direction == FacingDirection::WEST) {
211 pos_x -= distance;
212 is_moving = true;
213
214 if (this->direction != FacingDirection::WEST)
215 SetPlayerDirection(FacingDirection::WEST);
216 }
217
218 else if (direction == FacingDirection::EAST) {
219 pos_x += distance;
220 is_moving = true;
221
222 if (this->direction != FacingDirection::EAST)
223 SetPlayerDirection(FacingDirection::EAST);
224 }
225
226 if (is_moving) {
227 if (state != PlayerState::RUN)
228 SetPlayerState(PlayerState::RUN);
229 } else {
230 if (state != PlayerState::IDLE)
231 SetPlayerState(PlayerState::IDLE);
232 }
233}
