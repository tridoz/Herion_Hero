163void InputProcessor::Process() {
164 switch (this->event.type) {
165 case SDL_EVENT_QUIT:
166 this->running = false;
167 break;
168
169 case SDL_EVENT_KEY_DOWN:
170 if (event.key.scancode < MAX_SCANCODES) {
171 this->keys[event.key.scancode] = true;
172 process_key_down(event.key.scancode);
173 }
174 break;
175
176 case SDL_EVENT_KEY_UP:
177 if (event.key.scancode < MAX_SCANCODES) {
178 this->keys[event.key.scancode] = false;
179 process_key_up(event.key.scancode);
180 }
181 break;
182
183 case SDL_EVENT_MOUSE_BUTTON_DOWN:
184 if (event.button.button == SDL_BUTTON_LEFT) {
185 process_mouse_left_pressed();
186 } else if (event.button.button == SDL_BUTTON_RIGHT) {
187 process_mouse_right_pressed();
188 }
189 break;
190
191 case SDL_EVENT_MOUSE_BUTTON_UP:
192 break;
193
194 case SDL_EVENT_MOUSE_WHEEL:
195
196 if ( player->GetGameMode() == Player::GameMode::TEXTURE_SELECTION ) {
197 for ( const auto& [name, menu] : menus ) {
198 menu->SetMouseOffset( event.wheel.y);
199 }
200 }
201
202 break;
203
204 default:
205 break;
206 }
207}
