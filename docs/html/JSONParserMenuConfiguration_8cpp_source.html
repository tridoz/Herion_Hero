142JSONParser::menu_configuration::RowElementFields
143JSONParser::menu_configuration::GetRowElementFields(int num_row, int num_element)
144{
145
146 if (
147 !config_file.contains("rows") ||
148 config_file["rows"].size() < num_row ||
149 !config_file["rows"][num_row].contains("elements") ||
150 config_file["rows"][num_row]["elements"].size() < num_element
151 )
152 {
153 THROW_FILE_MALFORMED( config_file_path + "\t=> rows , n_row: " + std::to_string(num_row) + ", button_x_offset, n_elem:" + std::to_string(num_element) );
154 }
155
156 const auto& element = config_file["rows"][num_row]["elements"][num_element];
157
158 RowElementFields result;
159 result.type = element.at("type").get<std::string>();
160 result.text = element.at("text").get<std::string>();
161 result.id = element.at("id").get<std::string>();
162
163 if (element.contains("action")) {
164 result.action = element.at("action").get<std::string>();
165 }
166
167 return result;
