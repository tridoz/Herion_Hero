21std::string Logger::FormatTime(const std::time_t time_to_format) {
22 const std::tm *lt = std::localtime(&time_to_format);
23 std::stringstream ss;
24
25 ss << std::setfill('0')
26 << (lt->tm_year + 1900) << ":"
27 << std::setw(2) << (lt->tm_mon + 1) << ":"
28 << std::setw(2) << lt->tm_mday << " "
29 << std::setw(2) << lt->tm_hour << ":"
30 << std::setw(2) << lt->tm_min << ":"
31 << std::setw(2) << lt->tm_sec;
32
33 return ss.str();
