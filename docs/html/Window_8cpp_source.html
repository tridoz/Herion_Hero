170void Window::LoadCursors( const std::string &filename ) {
171
172 std::ifstream cursors_file( "../" + filename, std::ios::in );
173 std::string line;
174 std::stringstream ss;
175
176 while ( std::getline( cursors_file, line) ) {
177 ss << line << std::endl;
178 std::string cursor_name, cursor_path;
179 int hotX, hotY;
180 ss >> cursor_name >> cursor_path >> hotX >> hotY;
181
182 ss.clear();
183 SDL_Surface* cursor_surface = IMG_Load( cursor_path.c_str() );
184 Cursors[cursor_name] = SDL_CreateColorCursor( cursor_surface, hotX, hotY);
185 SDL_DestroySurface( cursor_surface );
186
187 }
188
189}
